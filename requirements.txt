# Discord Music Bot Dependencies

# Core Dependencies
# discord.py 2.5+ is required for Python 3.13+ (older versions import the
# `audioop` module, which was removed from the standard library in Python 3.13).
discord.py>=2.7.1
# YouTube breaks older extractors regularly: an out-of-date yt-dlp returns
# HTTP 403 on the media download. Keep this floor recent and rebuild often.
yt-dlp>=2026.10.26
python-dotenv==1.2.4
# Web server for the synced clip page (already pulled in by discord.py).
aiohttp>=3.14.5

# Supplies the Proof-of-Origin tokens YouTube demands from datacenter IPs.
# Without it yt-dlp reports `PO Token Providers: none` and downloads get
# challenged with "Sign in to confirm you're not a bot".
#
# Pinned on purpose: this must match the `pot-provider` image tag in
# docker-compose.yml. Bump both together, never one alone -- a version mismatch
# between plugin and server breaks token minting. This is also why the
# entrypoint auto-updates yt-dlp but deliberately leaves this alone.
bgutil-ytdlp-pot-provider==2.0.2

# Audio Processing
PyNaCl==1.6.2

# Required by discord.py 2.7+ for voice (Discord's DAVE end-to-end encryption).
davey>=0.1.7

# audioop was removed from Python 3.13; this backport restores it for voice.
audioop-lts>=0.2.2; python_version >= "3.13"

# Optional Audio Enhancement (improves performance but not required)
# PyOpus==0.9

# Optional: Development and Testing
# pytest==7.4.3
# black==23.11.0
